3.3 New File: tools/pkgbase/build/stage.mk
------------------------------------------

Full content:
```makefile
#-
# SPDX-License-Identifier: BSD-2-Clause
#
# Copyright (c) 2026 The FreeBSD Foundation
#
# Staging targets for pkgbase package builds
#

# Stage world for packaging
stage-packages-world: .PHONY
	@mkdir -p ${WSTAGEDIR}
	${_+_}@cd ${.CURDIR}; \
		${MAKE} DESTDIR=${WSTAGEDIR} -DNO_ROOT stageworld

# Stage kernel for packaging
stage-packages-kernel: .PHONY
	@mkdir -p ${KSTAGEDIR}
	${_+_}@cd ${.CURDIR}; \
		${MAKE} DESTDIR=${KSTAGEDIR} -DNO_ROOT stagekernel

# Stage source for packaging
stage-packages-source: .PHONY
	@mkdir -p ${SSTAGEDIR}

# Meta-target for all staging
stage-packages: .PHONY stage-packages-world stage-packages-kernel stage-packages-source
```
